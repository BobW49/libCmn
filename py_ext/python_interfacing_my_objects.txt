// ------------------------------------------------------------------
// Convert AStr → Python str
// ------------------------------------------------------------------
static PyObject* AStr_ToPython(AStr_DATA* pAStr) {
    if (!pAStr) Py_RETURN_NONE;
    const char* utf8 = AStr_getData(pAStr);
    Py_ssize_t len = AStr_getLength(pAStr);
    return PyUnicode_FromStringAndSize(utf8, len);
}

// ------------------------------------------------------------------
// NODEHASH_DATA → Python dict
// ------------------------------------------------------------------
static PyObject* NodeHash_ToPython(NODEHASH_DATA* pHash) {
    if (!pHash) Py_RETURN_NONE;

    PyObject* pyDict = PyDict_New();
    if (!pyDict) return NULL;

    // You need the iteration API for NODEHASH — common pattern in your libs:
    NODEARRAY_DATA* pKeys = NodeHash_Keys(pHash);  // or similar method
    if (pKeys) {
        uint32_t i, cnt = NodeArray_getSize(pKeys);
        for (i = 0; i < cnt; i++) {
            NODE_DATA* pKeyNode = NodeArray_Get(pKeys, i);
            if (!pKeyNode) continue;

            const char* key = AStr_CStringA(Node_getData(pKeyNode), NULL);
            NODE_DATA* pValueNode = NodeHash_Find(pHash, key);  // or use pair iterator

            PyObject* pyKey = PyUnicode_FromString(key);
            PyObject* pyVal = Node_ToPython(pValueNode);

            if (pyKey && pyVal) {
                PyDict_SetItem(pyDict, pyKey, pyVal);
            }
            Py_XDECREF(pyKey);
            Py_XDECREF(pyVal);
        }
        obj_Release(pKeys);
    }

    return pyDict;
}

// ------------------------------------------------------------------
// NODEARRAY_DATA → Python list
// ------------------------------------------------------------------
static PyObject* NodeArray_ToPython(NODEARRAY_DATA* pArray) {
    if (!pArray) Py_RETURN_NONE;

    uint32_t size = NodeArray_getSize(pArray);
    PyObject* pyList = PyList_New(size);
    if (!pyList) return NULL;

    for (uint32_t i = 0; i < size; i++) {
        NODE_DATA* pNode = NodeArray_Get(pArray, i);
        PyObject* pyItem = Node_ToPython(pNode);
        PyList_SET_ITEM(pyList, i, pyItem);   // steals reference
    }
    return pyList;
}

// ------------------------------------------------------------------
// Main recursive converter
// ------------------------------------------------------------------
static PyObject* Node_ToPython(NODE_DATA* pNode) {
    if (!pNode) Py_RETURN_NONE;

    const char* name = AStr_CStringA(Node_getData(pNode), NULL);  // or Node_getName()

    // Special nodes from parser
    if (strcmp(name, "hash") == 0) {
        return NodeHash_ToPython((NODEHASH_DATA*)Node_getData(pNode));
    }
    if (strcmp(name, "array") == 0) {
        return NodeArray_ToPython((NODEARRAY_DATA*)Node_getData(pNode));
    }
    if (strcmp(name, "string") == 0 || strcmp(name, "name") == 0) {
        return AStr_ToPython((AStr_DATA*)Node_getData(pNode));
    }
    if (strcmp(name, "integer") == 0) {
        AStr_DATA* pStr = (AStr_DATA*)Node_getData(pNode);
        return PyLong_FromString(AStr_getData(pStr), NULL, 10);
    }
    if (strcmp(name, "float") == 0) {
        AStr_DATA* pStr = (AStr_DATA*)Node_getData(pNode);
        return PyFloat_FromDouble(atof(AStr_getData(pStr)));
    }
    if (strcmp(name, "true") == 0)  Py_RETURN_TRUE;
    if (strcmp(name, "false") == 0) Py_RETURN_FALSE;
    if (strcmp(name, "null") == 0)  Py_RETURN_NONE;

    // Fallback: just the string value
    return AStr_ToPython((AStr_DATA*)Node_getData(pNode));
}

// ------------------------------------------------------------------
// Python-exposed function
// ------------------------------------------------------------------
static PyObject* py_hjson_parse(PyObject* self, PyObject* args) {
    const char* filepath;

    if (!PyArg_ParseTuple(args, "s", &filepath))
        return NULL;

    HJSON_DATA* pHjson = hjson_NewFromPath(NULL, filepath, 4);  // tabSize=4
    if (!pHjson) {
        PyErr_SetString(PyExc_RuntimeError, "Failed to create HJSON parser");
        return NULL;
    }

    NODE_DATA* pRoot = hjson_ParseFileHash(pHjson);
    if (!pRoot) {
        // Try value instead?
        pRoot = hjson_ParseFileValue(pHjson);
    }

    obj_Release(pHjson);

    if (!pRoot) {
        PyErr_SetString(PyExc_RuntimeError, "HJSON parse failed (check SrcErrors)");
        return NULL;
    }

    PyObject* pyResult = Node_ToPython(pRoot);
    obj_Release(pRoot);

    return pyResult;
}

```c

#include <Python.h>
#include "hjson.h"

// ================================================================
//  Python creation hooks (these will be called from your C hooks)
// ================================================================

static PyObject* py_create_hash(void* ctx) {
    return PyDict_New();
}

static PyObject* py_create_array(void* ctx) {
    return PyList_New(0);
}

static PyObject* py_create_string(void* ctx, const char* utf8, size_t len) {
    return PyUnicode_FromStringAndSize(utf8, len);
}

static PyObject* py_create_number(void* ctx, const char* numstr, bool isFloat) {
    if (isFloat)
        return PyFloat_FromDouble(atof(numstr));
    return PyLong_FromString(numstr, NULL, 10);
}

static PyObject* py_create_bool(void* ctx, bool value) {
    return Py_NewRef(value ? Py_True : Py_False);
}

    Py_RETURN_NONE;
}
```
